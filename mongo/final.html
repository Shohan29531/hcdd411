  • The UI uses dropdowns for view, update, and delete
This keeps the frontend closer to how real projects are organized: HTML for layout, JavaScript for behavior.

  18. }
  19. });
  21. // ── GET /students/:id ── return one student by ID ─────────────
  22. app.get("/students/:id", async (req, res) => {
  23. try {
  24. const student = await Student.findById(req.params.id);
  25. if (!student) return res.status(404).json({ error: "Not found" });
  26. res.json(student);
  27. } catch (err) {
  28. res.status(400).json({ error: err.message });
  29. }
  30. });
  32. // ── POST /students ── create a new student ────────────────────
  33. app.post("/students", async (req, res) => {
  34. try {
  35. const student = await Student.create(req.body);
  36. res.status(201).json(student);
  37. } catch (err) {
  38. res.status(400).json({ error: err.message });
  39. }
  40. });
  42. // ── PUT /students/:id ── update a student by ID ───────────────
  43. app.put("/students/:id", async (req, res) => {
  44. try {
  45. const student = await Student.findByIdAndUpdate(
  46. req.params.id,
  47. { $set: req.body },
  48. { new: true, runValidators: true }
  49. );
  51. if (!student) return res.status(404).json({ error: "Not found" });
  52. res.json(student);
  53. } catch (err) {
  54. res.status(400).json({ error: err.message });
  55. }
  56. });
  58. // ── DELETE /students/:id ── delete a student by ID ───────────
  59. app.delete("/students/:id", async (req, res) => {
  60. try {
  61. const student = await Student.findByIdAndDelete(req.params.id);
  62. if (!student) return res.status(404).json({ error: "Not found" });
  63. res.json({ message: "Deleted successfully", student });
  64. } catch (err) {
  65. res.status(400).json({ error: err.message });
  66. }
  67. });
  69. async function startServer() {
  70. await connectDB();

  72. console.log("Server running on http://localhost:3000");
  73. });
  74. }

New pieces

  • app.use(express.static(...)) serves the frontend
  • GET /students/:id supports the dropdown-based view/update/delete UI

What did not change

  • The same DB connection flow
  • The same Student model
  • The same REST route pattern students already know

Step 6 — Update a Student from a Dropdown

Important design choice

  • The user picks a student from a dropdown
  • Changing the dropdown fills the form with the current values
  • Only filled fields are sent in the payload
  • The backend still uses $set, so untouched fields stay untouched

Step 7 — Delete a Student from a Dropdown

Why this UI is cleaner

  • No visible MongoDB IDs
  • No row-level delete buttons to distract from the lesson
  • The same backend delete route still powers the action
This keeps the browser UI focused on how the frontend talks to the backend, not on database internals.
